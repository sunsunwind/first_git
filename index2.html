function adjustAndConvert(currency, amount) { const inputEl = document.getElementById(`${currency}Input`); let current = parseFloat(inputEl.value) || 0; current += amount; if (current < 0) current = 0; inputEl.value = current; convertToKRW(currency); // 버튼 클릭 시 즉시 진동 (사용자 제스처 안에 포함) if (navigator.vibrate) navigator.vibrate(30); } function copyResult(currency) { const inputVal = parseFloat(document.getElementById(`${currency}Input`).value); const resultText = document.getElementById(`${currency}Result`).textContent; const currObj = currencies.find(c => c.code === currency); if (!inputVal) return alert("금액을 입력해주세요!"); const textToCopy = `${inputVal.toLocaleString()}${currObj.name}는 원화로 ${resultText} 입니다.`; navigator.clipboard.writeText(textToCopy).then(() => { alert('복사되었습니다:\n' + textToCopy); // 복사 버튼 진동 if (navigator.vibrate) navigator.vibrate(50); }); } function saveAsImage(currency) { const section = document.getElementById(`${currency}-section`); html2canvas(section, { scale: 2 }).then(canvas => { const link = document.createElement('a'); link.download = `${currency}_변환결과.png`; link.href = canvas.toDataURL(); link.click(); // 이미지 저장 버튼 진동 if (navigator.vibrate) navigator.vibrate(50); }); }
